   245→   245→/** Resolves the location tier from validated criteria. Unknown IDs throw a
   246→   246→ *  ValidationError (not 404) — they are query parameters, not resources. */
   247→   247→async function resolveLocation(query: DiscoveryQuery): Promise<ResolvedLocation> {
   248→   248→  // Community → its town (service areas are Town-level).
   249→   249→  if (query.communityId) {
   250→   250→    const community = await db.communityArea.findFirst({
   251→   251→      where: { id: query.communityId, isActive: true },
   252→   252→      include: {
   253→   253→        town: { include: { district: true, region: true } },
   254→   254→      },
   255→   255→    })
   256→   256→    if (!community) throw new ValidationError({ communityId: ['Unknown community'] })
   257→   257→    const { town } = community
   258→   258→    return {
   259→   259→      level: 'community',
   260→   260→      point: pointOf(community.latitude, community.longitude) ?? pointOf(town.latitude, town.longitude),
   261→   261→      scope: { kind: 'town', townId: town.id },
   262→   262→      label: `${community.name} (${town.name})`,
   263→   263→      breadcrumb: [
   264→   264→        { label: 'Ghana', href: '/find' },
   265→   265→        { label: town.region.name, href: `/find?regionId=${town.regionId}` },
   266→   266→        { label: town.district.name, href: null },
   267→   267→        { label: town.name, href: `/find?townId=${town.id}` },
   268→   268→        { label: community.name, href: null },
   269→   269→      ],
   270→   270→      contextTownId: town.id,
   271→   271→    }
   272→   272→  }
   273→   273→
   274→   274→  if (query.townId) {
   275→   275→    const town = await db.town.findFirst({
   276→   276→      where: { id: query.townId, isActive: true },
   277→   277→      include: { district: true, region: true },
   278→   278→    })
   279→   279→    if (!town) throw new ValidationError({ townId: ['Unknown town'] })
   280→   280→    return {
   281→   281→      level: 'town',
   282→   282→      point: pointOf(town.latitude, town.longitude),
   283→   283→      scope: { kind: 'town', townId: town.id },
   284→   284→      label: town.name,
   285→   285→      breadcrumb: [
   286→   286→        { label: 'Ghana', href: '/find' },
   287→   287→        { label: town.region.name, href: `/find?regionId=${town.regionId}` },
   288→   288→        { label: town.district.name, href: null },
   289→   289→        { label: town.name, href: `/find?townId=${town.id}` },
   290→   290→      ],
   291→   291→      contextTownId: town.id,
   292→   292→    }
   293→   293→  }
   294→   294→
   295→   295→  if (query.districtId) {
   296→   296→    const district = await db.district.findFirst({
   297→   297→      where: { id: query.districtId, isActive: true },
   298→   298→      include: { region: true },
   299→   299→    })
   300→   300→    if (!district) throw new ValidationError({ districtId: ['Unknown district'] })
   301→   301→    const towns = await db.town.findMany({
   302→   302→      where: { districtId: district.id, isActive: true },
   303→   303→      select: { id: true },
   304→   304→    })
   305→   305→    return {
   306→   306→      level: 'district',
   307→   307→      point: pointOf(district.latitude, district.longitude),
   308→   308→      scope: { kind: 'district', districtId: district.id, townIds: towns.map((t) => t.id) },
   309→   309→      label: district.name,
   310→   310→      breadcrumb: [
   311→   311→        { label: 'Ghana', href: '/find' },
   312→   312→        { label: district.region.name, href: `/find?regionId=${district.regionId}` },
   313→   313→        { label: district.name, href: null },
   314→   314→      ],
   315→   315→      contextTownId: null,
   316→   316→    }
   317→   317→  }
   318→   318→
   319→   319→  if (query.regionId) {
   320→   320→    const region = await db.region.findFirst({
   321→   321→      where: { id: query.regionId, isActive: true },
   322→   322→    })
   323→   323→    if (!region) throw new ValidationError({ regionId: ['Unknown region'] })
   324→   324→    return {
   325→   325→      level: 'region',
   326→   326→      point: pointOf(region.latitude, region.longitude),
   327→   327→      scope: { kind: 'region', regionId: region.id },
   328→   328→      label: `${region.name} Region`,
   329→   329→      breadcrumb: [
   330→   330→        { label: 'Ghana', href: '/find' },
   331→   331→        { label: region.name, href: null },
   332→   332→      ],
   333→   333→      contextTownId: null,
   334→   334→    }
   335→   335→  }
   336→   336→
   337→   337→  if (query.latitude !== undefined && query.longitude !== undefined) {
   338→   338→    return {
   339→   339→      level: null,
   340→   340→      point: { latitude: query.latitude, longitude: query.longitude },
   341→   341→      scope: { kind: 'none' },
   342→   342→      label: null,
   343→   343→      breadcrumb: [{ label: 'Ghana', href: '/find' }],
   344→   344→      contextTownId: null,
   345→   345→    }
   346→   346→  }
   347→   347→
   348→   348→  return {
   349→   349→    level: null,
   350→   350→    point: null,
   351→   351→    scope: { kind: 'none' },
   352→   352→    label: null,
   353→   353→    breadcrumb: [{ label: 'Ghana', href: '/find' }],
   354→   354→    contextTownId: null,
   355→   355→  }
   356→   356→}

   368→   368→/** Resolves the requested category by ID or slug and expands its subtree. */
   369→   369→async function resolveCategory(query: DiscoveryQuery): Promise<ResolvedCategory | null> {
   370→   370→  const base = query.categoryId
   371→   371→    ? await db.category.findFirst({ where: { id: query.categoryId, isActive: true } })
   372→   372→    : query.categorySlug
   373→   373→      ? await db.category.findFirst({ where: { slug: query.categorySlug, isActive: true } })
   374→   374→      : null
   375→   375→  if (!base) {
   376→   376→    if (query.categoryId || query.categorySlug) {
   377→   377→      throw new ValidationError(
   378→   378→        query.categoryId
   379→   379→          ? { categoryId: ['Unknown category'] }
   380→   380→          : { categorySlug: ['Unknown category'] },
   381→   381→      )
   382→   382→    }
   383→   383→    return null
   384→   384→  }
   385→   385→
   386→   386→  // Subtree expansion (depth ≤ 3): children, then grandchildren.
   387→   387→  const subtreeIds = [base.id]
   388→   388→  let frontier = [base.id]
   389→   389→  for (let depth = 0; depth < 2 && frontier.length > 0; depth += 1) {
   390→   390→    const children = await db.category.findMany({
   391→   391→      where: { parentId: { in: frontier }, isActive: true },
   392→   392→      select: { id: true },
   393→   393→    })
   394→   394→    frontier = children.map((c) => c.id)
   395→   395→    subtreeIds.push(...frontier)
   396→   396→  }
   397→   397→
   398→   398→  return {
   399→   399→    id: base.id,
   400→   400→    name: base.name,
   401→   401→    slug: base.slug,
   402→   402→    description: base.description,
   403→   403→    subtreeIds,
   404→   404→  }
   405→   405→}

   446→   446→/** Builds the hard-filter WHERE clause for a resolved scope. */
   447→   447→function locationWhere(scope: ResolvedLocation['scope']) {
   448→   448→  switch (scope.kind) {
   449→   449→    case 'town':
   450→   450→      return {
   451→   451→        OR: [
   452→   452→          { primaryLocationId: scope.townId },
   453→   453→          { serviceAreas: { some: { locationId: scope.townId } } },
   454→   454→        ],
   455→   455→      }
   456→   456→    case 'district':
   457→   457→      return {
   458→   458→        OR: [
   459→   459→          { primaryLocationId: { in: scope.townIds } },
   460→   460→          { serviceAreas: { some: { locationId: { in: scope.townIds } } } },
   461→   461→        ],
   462→   462→      }
   463→   463→    case 'region':
   464→   464→      return {
   465→   465→        OR: [
   466→   466→          { primaryLocation: { regionId: scope.regionId } },
   467→   467→          { serviceAreas: { some: { location: { regionId: scope.regionId } } } },
   468→   468→        ],
   469→   469→      }
   470→   470→    default:
   471→   471→      return {}
   472→   472→  }

   484→   484→  ratingSum: true,
   485→   485→  ratingCount: true,
   486→   486→  completedJobsCount: true,
   487→   487→  responseRatePercent: true,
   488→   488→  businessId: true,
   489→   489→  user: { select: { name: true, avatarKey: true } },
   490→   490→  business: { select: { id: true, name: true } },
   491→   491→  primaryLocation: {
   492→   492→    select: { id: true, name: true, latitude: true, longitude: true, region: { select: { name: true } } },
   493→   493→  },
   494→   494→  serviceAreas: {
   495→   495→    select: {
   496→   496→      location: {
   497→   497→        select: { id: true, name: true, latitude: true, longitude: true, region: { select: { name: true } } },
   498→   498→      },
   499→   499→    },
   500→   500→  },
   501→   501→  _count: { select: { portfolioItems: true } },
   502→   502→} as const
   503→   503→
   504→   504→function shapeCandidate(row: {
   505→   505→  id: string
   506→   506→  profession: string
   507→   507→  headline: string | null
   508→   508→  yearsExperience: number
   509→   509→  availabilityStatus: string
   510→   510→  startingPriceAmount: number | null
   511→   511→  currency: string
   512→   512→  verificationStatus: string
   513→   513→  ratingSum: number
   514→   514→  ratingCount: number
   515→   515→  completedJobsCount: number
   516→   516→  responseRatePercent: number | null
   517→   517→  businessId: string | null
   518→   518→  user: { name: string | null; avatarKey: string | null }
   519→   519→  business: { id: string; name: string } | null
   520→   520→  primaryLocation: {
   521→   521→    id: string
   522→   522→    name: string
   523→   523→    latitude: number | null
   524→   524→    longitude: number | null
   525→   525→    region: { name: string } | null
   526→   526→  } | null
   527→   527→  serviceAreas: {
   528→   528→    location: {
   529→   529→      id: string
   530→   530→      name: string
   531→   531→      latitude: number | null
   532→   532→      longitude: number | null
   533→   533→      region: { name: string } | null
   534→   534→    }
   535→   535→  }[]
   536→   536→  _count: { portfolioItems: number }
   537→   537→}): Candidate {
   538→   538→  return {
   539→   539→    id: row.id,
   540→   540→    profession: row.profession,
   541→   541→    headline: row.headline,
   542→   542→    yearsExperience: row.yearsExperience,
   543→   543→    availabilityStatus: row.availabilityStatus,
   544→   544→    startingPriceAmount: row.startingPriceAmount,
   545→   545→    currency: row.currency,
   546→   546→    verificationStatus: row.verificationStatus,
   547→   547→    ratingSum: row.ratingSum,
   548→   548→    ratingCount: row.ratingCount,
   549→   549→    completedJobsCount: row.completedJobsCount,
   550→   550→    responseRatePercent: row.responseRatePercent,
   551→   551→    businessId: row.businessId,
   552→   552→    user: row.user,
   553→   553→    business: row.business,
   554→   554→    primaryLocation: row.primaryLocation
   555→   555→      ? {
   556→   556→          id: row.primaryLocation.id,
   557→   557→          name: row.primaryLocation.name,
   558→   558→          regionName: row.primaryLocation.region?.name ?? null,
   559→   559→          latitude: row.primaryLocation.latitude,
   560→   560→          longitude: row.primaryLocation.longitude,
   561→   561→        }
   562→   562→      : null,
   563→   563→    serviceAreas: row.serviceAreas.map(({ location }) => ({
   564→   564→      location: {
   565→   565→        id: location.id,
   566→   566→        name: location.name,
   567→   567→        regionName: location.region?.name ?? null,
   568→   568→        latitude: location.latitude,
   569→   569→        longitude: location.longitude,
   570→   570→      },
   571→   571→    })),
   572→   572→    matchedServices: [],
   573→   573→    portfolioCount: row._count.portfolioItems,
   574→   574→  }
   575→   575→}

   577→   577→// -----------------------------------------------------------------------------
   578→   578→// Scoring (PART 10 — weights from the central configuration only)
   579→   579→// -----------------------------------------------------------------------------
   580→   580→
   581→   581→interface ScoredComponent {
   582→   582→  key: keyof typeof DISCOVERY_WEIGHTS
   583→   583→  score: number
   584→   584→  available: boolean
   585→   585→}
   586→   586→
   587→   587→interface ScoredCandidate {
   588→   588→  candidate: Candidate
   589→   589→  components: Record<string, ScoredComponent>
   590→   590→  total: number // 0–100
   591→   591→  distanceKm: number | null
   592→   592→  distanceBand: DistanceBand | null
   593→   593→  exactLocation: boolean
   594→   594→  reasons: string[]
   595→   595→}
   596→   596→
   597→   597→function scoreCandidate(
   598→   598→  candidate: Candidate,
   599→   599→  context: {
   600→   600→    category: ResolvedCategory | null
   601→   601→    location: ResolvedLocation
   602→   602→    maxDistanceKm: number | null
   603→   603→  },
   604→   604→): ScoredCandidate {
   605→   605→  const reasons: string[] = []
   606→   606→
   607→   607→  // SERVICE MATCH ------------------------------------------------------------
   608→   608→  let serviceScore: ScoredComponent = {
   609→   609→    key: 'serviceMatch',
   610→   610→    score: 0,
   611→   611→    available: context.category !== null,
   612→   612→  }
   613→   613→  if (context.category) {
   614→   614→    const exact = candidate.matchedServices.some((s) => s.categoryId === context.category!.id)
   615→   615→    serviceScore = { ...serviceScore, score: exact ? SERVICE_MATCH_SCORES.exact : SERVICE_MATCH_SCORES.ancestor }
   616→   616→    if (exact) reasons.push(`Offers ${context.category.name}`)
   617→   617→  }
   618→   618→
   619→   619→  // LOCATION / SERVICE AREA MATCH -------------------------------------------
   620→   620→  const exactTownId =
   621→   621→    context.location.scope.kind === 'town'
   622→   622→      ? context.location.scope.townId
   623→   623→      : context.location.contextTownId
   624→   624→  const exactLocation =
   625→   625→    exactTownId !== null &&
   626→   626→    (candidate.primaryLocation?.id === exactTownId ||
   627→   627→      candidate.serviceAreas.some((area) => area.location.id === exactTownId))
   628→   628→
   629→   629→  let distanceKm: number | null = null
   630→   630→  if (context.location.point) {
   631→   631→    const towns = [
   632→   632→      candidate.primaryLocation,
   633→   633→      ...candidate.serviceAreas.map((a) => a.location),
   634→   634→    ].filter((town): town is CandidateTown => town !== null)
   635→   635→    let best: number | null = null
   636→   636→    for (const town of towns) {
   637→   637→      const townPoint = pointOf(town.latitude, town.longitude)
   638→   638→      if (!townPoint) continue
   639→   639→      const d = haversineKmLocal(context.location.point, townPoint)
   640→   640→      if (best === null || d < best) best = d
   641→   641→    }
   642→   642→    distanceKm = best
   643→   643→  }
   644→   644→
   645→   645→  let locationScore: number
   646→   646→  if (exactLocation) {
   647→   647→    locationScore = LOCATION_HIERARCHY_SCORES.communityOrTown
   648→   648→    reasons.push('Serves your area')
   649→   649→  } else if (distanceKm !== null) {
   650→   650→    const band = distanceBandFor(distanceKm)
   651→   651→    locationScore = band.score
   652→   652→    reasons.push(`About ${band.label} away`)
   653→   653→  } else {
   654→   654→    locationScore =
   655→   655→      context.location.scope.kind === 'district'
   656→   656→        ? LOCATION_HIERARCHY_SCORES.districtSweep
   657→   657→        : LOCATION_HIERARCHY_SCORES.regionSweep
   658→   658→    reasons.push('Serves your wider area')
   659→   659→  }
   660→   660→  const locationComponent: ScoredComponent = {
   661→   661→    key: 'locationMatch',
   662→   662→    score: locationScore,
   663→   663→    available: context.location.scope.kind !== 'none' || context.location.point !== null,
   664→   664→  }
   665→   665→
   666→   666→  // AVAILABILITY (real field, PART 13 — never inferred from account status) --
   667→   667→  const availabilityScore = AVAILABILITY_SCORES[candidate.availabilityStatus] ?? NEUTRAL_COMPONENT_SCORE
   668→   668→  if (candidate.availabilityStatus === 'AVAILABLE') reasons.push('Available now')
   669→   669→  const availabilityComponent: ScoredComponent = {
   670→   670→    key: 'availability',
   671→   671→    score: availabilityScore,
   672→   672→    available: true,
   673→   673→  }
   674→   674→
   675→   675→  // RATING / REVIEW QUALITY --------------------------------------------------
   676→   676→  const hasRating = candidate.ratingCount >= RATING_MIN_COUNT
   677→   677→  const ratingAverage = hasRating ? candidate.ratingSum / candidate.ratingCount : null
   678→   678→  const ratingComponent: ScoredComponent = {
   679→   679→    key: 'rating',
   680→   680→    score: hasRating ? Math.min(Math.max(ratingAverage! / 5, 0), 1) : NEUTRAL_COMPONENT_SCORE,
   681→   681→    available: hasRating, // unrated providers are NOT falsely scored
   682→   682→  }
   683→   683→  if (hasRating && ratingAverage! >= 4) reasons.push('Highly rated')
   684→   684→
   685→   685→  // RESPONSE RATE ------------------------------------------------------------
   686→   686→  const responseComponent: ScoredComponent = {
   687→   687→    key: 'responseRate',
   688→   688→    score:
   689→   689→      candidate.responseRatePercent === null
   690→   690→        ? NEUTRAL_COMPONENT_SCORE
   691→   691→        : Math.min(Math.max(candidate.responseRatePercent / 100, 0), 1),
   692→   692→    available: candidate.responseRatePercent !== null,
   693→   693→  }
   694→   694→
   695→   695→  // EXPERIENCE ---------------------------------------------------------------
   696→   696→  const experienceComponent: ScoredComponent = {
   697→   697→    key: 'experience',
   698→   698→    score: Math.min(candidate.yearsExperience / EXPERIENCE_FULL_SCORE_YEARS, 1),
   699→   699→    available: candidate.yearsExperience > 0,
   700→   700→  }
   701→   701→
   702→   702→  // VERIFICATION -------------------------------------------------------------
   703→   703→  const verificationScore = VERIFICATION_SCORES[candidate.verificationStatus] ?? 0
   704→   704→  if (candidate.verificationStatus === 'VERIFIED') reasons.push('Verified provider')
   705→   705→  const verificationComponent: ScoredComponent = {
   706→   706→    key: 'verification',
   707→   707→    score: verificationScore,
   708→   708→    available: true,
   709→   709→  }
   710→   710→
   711→   711→  const components: Record<string, ScoredComponent> = {
   712→   712→    serviceMatch: serviceScore,
   713→   713→    locationMatch: locationComponent,
   714→   714→    availability: availabilityComponent,
   715→   715→    rating: ratingComponent,
   716→   716→    responseRate: responseComponent,
   717→   717→    experience: experienceComponent,
   718→   718→    verification: verificationComponent,
   719→   719→  }
   720→   720→
   721→   721→  // Neutral-fill: components without real data score NEUTRAL (0.5) and keep
   722→   722→  // their weight — missing information pulls toward the middle, never boosts
   723→   723→  // an empty profile above a substantive one, and never fabricates data.
   724→   724→  let weighted = 0
   725→   725→  for (const component of Object.values(components)) {
   726→   726→    const score = component.available ? component.score : NEUTRAL_COMPONENT_SCORE
   727→   727→    weighted += DISCOVERY_WEIGHTS[component.key] * score
   728→   728→  }
   729→   729→  const total = weighted * 100
   730→   730→
   731→   731→  return {
   732→   732→    candidate,
   733→   733→    components,
   734→   734→    total,
   735→   735→    distanceKm,
   736→   736→    distanceBand: distanceKm !== null ? distanceBandFor(distanceKm) : null,
   737→   737→    exactLocation,
   738→   738→    reasons: [...new Set(reasons)],
   739→   739→  }
   740→   740→}
   741→   741→
   742→   742→// -----------------------------------------------------------------------------
   743→   743→// Sorting (PART 15 — unavailable information never falsely ranks)
   744→   744→// -----------------------------------------------------------------------------
   745→   745→
   746→   746→function sortScored(scored: ScoredCandidate[], sort: DiscoveryQuery['sort']): ScoredCandidate[] {
   747→   747→  const byScore = (a: ScoredCandidate, b: ScoredCandidate) => b.total - a.total || a.candidate.id.localeCompare(b.candidate.id)
   748→   748→  const list = [...scored]
   749→   749→
   750→   750→  switch (sort) {
   751→   751→    case 'nearest':
   752→   752→      return list.sort((a, b) => {
   753→   753→        if (a.distanceKm === null && b.distanceKm === null) return byScore(a, b)
   754→   754→        if (a.distanceKm === null) return 1
   755→   755→        if (b.distanceKm === null) return -1
   756→   756→        return a.distanceKm - b.distanceKm || byScore(a, b)
   757→   757→      })
   758→   758→    case 'rating':
   759→   759→      return list.sort((a, b) => {
   760→   760→        const aRated = a.candidate.ratingCount >= RATING_MIN_COUNT
   761→   761→        const bRated = b.candidate.ratingCount >= RATING_MIN_COUNT
   762→   762→        if (aRated !== bRated) return aRated ? -1 : 1
   763→   763→        if (aRated && bRated) {
   764→   764→          const aAvg = a.candidate.ratingSum / a.candidate.ratingCount
   765→   765→          const bAvg = b.candidate.ratingSum / b.candidate.ratingCount
   766→   766→          if (bAvg !== aAvg) return bAvg - aAvg
   767→   767→        }
   768→   768→        return byScore(a, b)
   769→   769→      })
   770→   770→    case 'experience':
   771→   771→      return list.sort(
   772→   772→        (a, b) => b.candidate.yearsExperience - a.candidate.yearsExperience || byScore(a, b),
   773→   773→      )
   774→   774→    case 'price':
   775→   775→      return list.sort((a, b) => {
   776→   776→        if (a.candidate.startingPriceAmount === null && b.candidate.startingPriceAmount === null) return byScore(a, b)
   777→   777→        if (a.candidate.startingPriceAmount === null) return 1
   778→   778→        if (b.candidate.startingPriceAmount === null) return -1
   779→   779→        return a.candidate.startingPriceAmount - b.candidate.startingPriceAmount || byScore(a, b)
   780→   780→      })
   781→   781→    case 'response':
   782→   782→      return list.sort((a, b) => {
   783→   783→        if (a.candidate.responseRatePercent === null && b.candidate.responseRatePercent === null) return byScore(a, b)
   784→   784→        if (a.candidate.responseRatePercent === null) return 1
   785→   785→        if (b.candidate.responseRatePercent === null) return -1
   786→   786→        return b.candidate.responseRatePercent - a.candidate.responseRatePercent || byScore(a, b)
   787→   787→      })
   788→   788→    default:
   789→   789→      return list.sort(byScore)
   790→   790→  }
   791→   791→}

   794→   794→// The pipeline (PART 20)

   796→   796→
   797→   797→export async function discoverProviders(query: DiscoveryQuery): Promise<DiscoveryResponse> {
   798→   798→  const [location, category] = await Promise.all([resolveLocation(query), resolveCategory(query)])
   799→   799→
   800→   800→  // Service search implies the service's own category for scoring purposes.
   801→   801→  let serviceId: string | null = null
   802→   802→  let effectiveCategory = category
   803→   803→  if (query.serviceId) {
   804→   804→    const service = await db.service.findFirst({
   805→   805→      where: { id: query.serviceId, status: 'ACTIVE', deletedAt: null },
   806→   806→      select: { id: true, categoryId: true, category: { select: { id: true, name: true, slug: true } } },
   807→   807→    })
   808→   808→    if (!service) throw new ValidationError({ serviceId: ['Unknown service'] })
   809→   809→    serviceId = service.id
   810→   810→    if (!effectiveCategory) {
   811→   811→      effectiveCategory = {
   812→   812→        id: service.category.id,
   813→   813→        name: service.category.name,
   814→   814→        slug: service.category.slug,
   815→   815→        description: null,
   816→   816→        subtreeIds: [service.category.id],
   817→   817→      }
   818→   818→    }
   819→   819→  }
   820→   820→
   821→   821→  // ---- Hard filters (PART 10) ---------------------------------------------
   822→   822→  // Conditions are composed as an AND list so independent OR-shapes (location
   823→   823→  // scope vs free-text q) can never collide on a top-level `OR` key.
   824→   824→  const baseConditions: Record<string, unknown>[] = [
   825→   825→    // Suspension is a LIVE user-status check (PART 45) — suspended providers
   826→   826→    // never appear publicly, even mid-session.
   827→   827→    { user: { status: 'ACTIVE' } },
   828→   828→  ]
   829→   829→  if (query.verification) baseConditions.push({ verificationStatus: query.verification })
   830→   830→  if (query.availability) baseConditions.push({ availabilityStatus: query.availability })
   831→   831→  if (query.providerType === 'INDIVIDUAL') baseConditions.push({ businessId: null })
   832→   832→  if (query.providerType === 'BUSINESS') baseConditions.push({ businessId: { not: null } })
   833→   833→  if (query.maxPrice !== undefined) {
   834→   834→    baseConditions.push({ startingPriceAmount: { lte: query.maxPrice, not: null } })
   835→   835→  }
   836→   836→  if (effectiveCategory) {
   837→   837→    baseConditions.push({
   838→   838→      services: {
   839→   839→        some: {
   840→   840→          status: 'ACTIVE',
   841→   841→          deletedAt: null,
   842→   842→          isAvailable: true,
   843→   843→          categoryId: { in: effectiveCategory.subtreeIds },
   844→   844→          ...(serviceId ? { id: serviceId } : {}),
   845→   845→        },
   846→   846→      },
   847→   847→    })
   848→   848→  }
   849→   849→  if (query.q) {
   850→   850→    baseConditions.push({
   851→   851→      OR: [
   852→   852→        { profession: { contains: query.q.toLowerCase() } },
   853→   853→        { headline: { contains: query.q } },
   854→   854→        { biography: { contains: query.q } },
   855→   855→        { business: { name: { contains: query.q } } },
   856→   856→      ],
   857→   857→    })
   858→   858→  }
   859→   859→  const locationCondition = locationWhere(location.scope)
   860→   860→  // `where` = base hard filters + the exact location tier. `baseConditions`
   861→   861→  // itself stays location-free: the nearby fallback reuses it with a swapped
   862→   862→  // location condition (requested town → nearby towns).
   863→   863→  const where = {
   864→   864→    deletedAt: null,
   865→   865→    AND:
   866→   866→      Object.keys(locationCondition).length > 0
   867→   867→        ? [...baseConditions, locationCondition]
   868→   868→        : baseConditions,
   869→   869→  }
   870→   870→  const serviceSelect = {
   871→   871→    where: {
   872→   872→      status: 'ACTIVE' as const,
   873→   873→      deletedAt: null as null,
   874→   874→      ...(effectiveCategory ? { categoryId: { in: effectiveCategory.subtreeIds } } : {}),
   875→   875→    },
   876→   876→    select: { name: true, categoryId: true },
   877→   877→    take: 10,
   878→   878→  }
   879→   879→
   880→   880→  // ---- Fetch candidates (single query, capped, minimal selects) ------------
   881→   881→  const rows = await db.providerProfile.findMany({
   882→   882→    where,
   883→   883→    take: DISCOVERY_CANDIDATE_CAP,
   884→   884→    select: { ...CANDIDATE_SELECT, services: serviceSelect },
   885→   885→  })
   886→   886→
   887→   887→  const candidates: Candidate[] = rows.map((row) => {
   888→   888→    const shaped = shapeCandidate(row)
   889→   889→    shaped.matchedServices = row.services
   890→   890→    return shaped
   891→   891→  })
   892→   892→
   893→   893→  // Post-fetch hard filters that SQL cannot express on SQLite (rating).
   894→   894→  const minRating = query.minRating ?? null
   895→   895→  const eligible = candidates.filter((candidate) => {
   896→   896→    if (minRating !== null && minRating > 0) {
   897→   897→      if (candidate.ratingCount < RATING_MIN_COUNT) return false
   898→   898→      if (candidate.ratingSum / candidate.ratingCount < minRating) return false
   899→   899→    }
   900→   900→    return true
   901→   901→  })
   902→   902→
   903→   903→  // ---- Score, filter by distance, sort, paginate ---------------------------
   904→   904→  const scoringContext = { category: effectiveCategory, location, maxDistanceKm: query.maxDistanceKm ?? null }
   905→   905→  let scored = eligible
   906→   906→    .map((candidate) => scoreCandidate(candidate, scoringContext))
   907→   907→    .filter((s) => {
   908→   908→      if (query.maxDistanceKm !== undefined && s.distanceKm !== null) {
   909→   909→        return s.distanceKm <= query.maxDistanceKm
   910→   910→      }
   911→   911→      return true
   912→   912→    })
   913→   913→
   914→   914→  // Duplicate prevention is structural: one row per ProviderProfile.
   915→   915→  const seen = new Set<string>()
   916→   916→  scored = scored.filter((s) => (seen.has(s.candidate.id) ? false : (seen.add(s.candidate.id), true)))
   917→   917→
   918→   918→  const ranked = sortScored(scored, query.sort)
   919→   919→  const total = ranked.length
   920→   920→  const page = query.page
   921→   921→  const pageSize = query.pageSize
   922→   922→  const pageItems = ranked.slice((page - 1) * pageSize, (page - 1) * pageSize + pageSize)
   923→   923→
   924→   924→  // ---- Nearby fallback (PART 18) — only when nothing matched exactly and a
   925→   925→  // real search point exists to measure "nearby" from --------------------------
   926→   926→  let nearby: DiscoveryResponse['nearby'] = null
   927→   927→  if (total === 0 && location.point) {
   928→   928→    const nearbyTowns = await db.town.findMany({
   929→   929→      where: {
   930→   930→        isActive: true,
   931→   931→        latitude: { not: null },
   932→   932→        longitude: { not: null },
   933→   933→      },
   934→   934→      select: { id: true, name: true, latitude: true, longitude: true },
   935→   935→    })
   936→   936→    const within = nearbyTowns
   937→   937→      .filter((town) => {
   938→   938→        const d = haversineKmLocal(
   939→   939→          location.point!,
   940→   940→          { latitude: town.latitude as number, longitude: town.longitude as number },
   941→   941→        )

  1029→  1029→/** Suggested nearby areas for empty states: same-district towns, then towns

  1032→  1032→  if (location.scope.kind === 'town') {
  1033→  1033→    const town = await db.town.findFirst({

  1038→  1038→

  1073→  1073→    const towns = await db.town.findMany({

  1075→  1075→      select: { id: true, name: true, slug: true },

  1078→  1078→    })

  1082→  1082→  return []
  1083→  1083→}
