     1→/**
     2→ * Dwellers — Provider discovery engine (Phase 4).
     3→ *
     4→ * THE one discovery backend for every entry point (homepage search, /find,
     5→ * category landings, location landings, provider directory, search URLs and
     6→ * the REST API). Nothing in the UI re-implements search logic (PART 2).
     7→ *
     8→ * Pipeline (PART 20):
     9→ *   1. validate criteria (zod, strict — no arbitrary query values)
    10→ *   2. resolve location (community→town, district/region sweeps, coordinates)
    11→ *   3. resolve service/category (id or slug, subtree expansion)
    12→ *   4. hard filters — active account, offered service, covering service area,
    13→ *      public eligibility (suspension is a LIVE user-status check)
    14→ *   5. fetch eligible candidates (capped, minimal selects, no N+1)
    15→ *   6. score relevance via the central ranking configuration
    16→ *   7. rank, 8. sort, 9. paginate (server-side), 10. return a standardized,
    17→ *      privacy-safe result envelope (no emails, phones or raw coordinates).
    18→ *
    19→ * Matching is grounded in real rows: "provider offers the service" means an
    20→ * ACTIVE Service row inside the requested category subtree; "serves the
    21→ * area" means a ProviderServiceArea row (or the profile base location).
    22→ * Nothing is inferred from free-text addresses (PART 11).
    23→ */

   245→/** Resolves the location tier from validated criteria. Unknown IDs throw a
   246→ *  ValidationError (not 404) — they are query parameters, not resources. */
   247→async function resolveLocation(query: DiscoveryQuery): Promise<ResolvedLocation> {
   248→  // Community → its town (service areas are Town-level).
   249→  if (query.communityId) {
   250→    const community = await db.communityArea.findFirst({
   251→      where: { id: query.communityId, isActive: true },
   252→      include: {
   253→        town: { include: { district: true, region: true } },
   254→      },
   255→    })
   256→    if (!community) throw new ValidationError({ communityId: ['Unknown community'] })
   257→    const { town } = community
   258→    return {
   259→      level: 'community',
   260→      point: pointOf(community.latitude, community.longitude) ?? pointOf(town.latitude, town.longitude),
   261→      scope: { kind: 'town', townId: town.id },
   262→      label: `${community.name} (${town.name})`,
   263→      breadcrumb: [
   264→        { label: 'Ghana', href: '/find' },
   265→        { label: town.region.name, href: `/find?regionId=${town.regionId}` },
   266→        { label: town.district.name, href: null },
   267→        { label: town.name, href: `/find?townId=${town.id}` },
   268→        { label: community.name, href: null },
   269→      ],
   270→      contextTownId: town.id,
   271→    }
   272→  }
   273→
   274→  if (query.townId) {
   275→    const town = await db.town.findFirst({
   276→      where: { id: query.townId, isActive: true },
   277→      include: { district: true, region: true },
   278→    })
   279→    if (!town) throw new ValidationError({ townId: ['Unknown town'] })
   280→    return {
   281→      level: 'town',
   282→      point: pointOf(town.latitude, town.longitude),
   283→      scope: { kind: 'town', townId: town.id },
   284→      label: town.name,
   285→      breadcrumb: [
   286→        { label: 'Ghana', href: '/find' },
   287→        { label: town.region.name, href: `/find?regionId=${town.regionId}` },
   288→        { label: town.district.name, href: null },
   289→        { label: town.name, href: `/find?townId=${town.id}` },
   290→      ],
   291→      contextTownId: town.id,
   292→    }
   293→  }
   294→
   295→  if (query.districtId) {
   296→    const district = await db.district.findFirst({
   297→      where: { id: query.districtId, isActive: true },
   298→      include: { region: true },
   299→    })
   300→    if (!district) throw new ValidationError({ districtId: ['Unknown district'] })
   301→    const towns = await db.town.findMany({
   302→      where: { districtId: district.id, isActive: true },
   303→      select: { id: true },
   304→    })
   305→    return {
   306→      level: 'district',
   307→      point: pointOf(district.latitude, district.longitude),
   308→      scope: { kind: 'district', districtId: district.id, townIds: towns.map((t) => t.id) },
   309→      label: district.name,
   310→      breadcrumb: [
   311→        { label: 'Ghana', href: '/find' },
   312→        { label: district.region.name, href: `/find?regionId=${district.regionId}` },
   313→        { label: district.name, href: null },
   314→      ],
   315→      contextTownId: null,
   316→    }
   317→  }
   318→
   319→  if (query.regionId) {
   320→    const region = await db.region.findFirst({
   321→      where: { id: query.regionId, isActive: true },
   322→    })
   323→    if (!region) throw new ValidationError({ regionId: ['Unknown region'] })
   324→    return {
   325→      level: 'region',
   326→      point: pointOf(region.latitude, region.longitude),
   327→      scope: { kind: 'region', regionId: region.id },
   328→      label: `${region.name} Region`,
   329→      breadcrumb: [
   330→        { label: 'Ghana', href: '/find' },
   331→        { label: region.name, href: null },
   332→      ],
   333→      contextTownId: null,
   334→    }
   335→  }
   336→
   337→  if (query.latitude !== undefined && query.longitude !== undefined) {
   338→    return {
   339→      level: null,
   340→      point: { latitude: query.latitude, longitude: query.longitude },
   341→      scope: { kind: 'none' },
   342→      label: null,
   343→      breadcrumb: [{ label: 'Ghana', href: '/find' }],
   344→      contextTownId: null,
   345→    }
   346→  }
   347→
   348→  return {
   349→    level: null,
   350→    point: null,
   351→    scope: { kind: 'none' },
   352→    label: null,
   353→    breadcrumb: [{ label: 'Ghana', href: '/find' }],
   354→    contextTownId: null,
   355→  }
   356→}

   358→function pointOf(
   359→  latitude: number | null | undefined,
   360→  longitude: number | null | undefined,
   361→): { latitude: number; longitude: number } | null {
   362→  if (latitude === null || latitude === undefined || longitude === null || longitude === undefined) {
   363→    return null
   364→  }
   365→  return { latitude, longitude }
   366→}

   368→/** Resolves the requested category by ID or slug and expands its subtree. */
   369→async function resolveCategory(query: DiscoveryQuery): Promise<ResolvedCategory | null> {
   370→  const base = query.categoryId
   371→    ? await db.category.findFirst({ where: { id: query.categoryId, isActive: true } })
   372→    : query.categorySlug
   373→      ? await db.category.findFirst({ where: { slug: query.categorySlug, isActive: true } })
   374→      : null
   375→  if (!base) {
   376→    if (query.categoryId || query.categorySlug) {
   377→      throw new ValidationError(
   378→        query.categoryId
   379→          ? { categoryId: ['Unknown category'] }
   380→          : { categorySlug: ['Unknown category'] },
   381→      )
   382→    }
   383→    return null
   384→  }
   385→
   386→  // Subtree expansion (depth ≤ 3): children, then grandchildren.
   387→  const subtreeIds = [base.id]
   388→  let frontier = [base.id]
   389→  for (let depth = 0; depth < 2 && frontier.length > 0; depth += 1) {
   390→    const children = await db.category.findMany({
   391→      where: { parentId: { in: frontier }, isActive: true },
   392→      select: { id: true },
   393→    })
   394→    frontier = children.map((c) => c.id)
   395→    subtreeIds.push(...frontier)
   396→  }
   397→
   398→  return {
   399→    id: base.id,
   400→    name: base.name,
   401→    slug: base.slug,
   402→    description: base.description,
   403→    subtreeIds,
   404→  }
   405→}

   446→/** Builds the hard-filter WHERE clause for a resolved scope. */
   447→function locationWhere(scope: ResolvedLocation['scope']) {
   448→  switch (scope.kind) {
   449→    case 'town':
   450→      return {
   451→        OR: [
   452→          { primaryLocationId: scope.townId },
   453→          { serviceAreas: { some: { locationId: scope.townId } } },
   454→        ],
   455→      }
   456→    case 'district':
   457→      return {
   458→        OR: [
   459→          { primaryLocationId: { in: scope.townIds } },
   460→          { serviceAreas: { some: { locationId: { in: scope.townIds } } } },
   461→        ],
   462→      }
   463→    case 'region':
   464→      return {
   465→        OR: [
   466→          { primaryLocation: { regionId: scope.regionId } },
   467→          { serviceAreas: { some: { location: { regionId: scope.regionId } } } },
   468→        ],
   469→      }
   470→    default:
   471→      return {}
   472→  }
   473→}

   600→    category: ResolvedCategory | null
   601→    location: ResolvedLocation
   602→    maxDistanceKm: number | null
   603→  },
   604→): ScoredCandidate {
   605→  const reasons: string[] = []
   606→
   607→  // SERVICE MATCH ------------------------------------------------------------
   608→  let serviceScore: ScoredComponent = {
   609→    key: 'serviceMatch',
   610→    score: 0,
   611→    available: context.category !== null,
   612→  }
   613→  if (context.category) {
   614→    const exact = candidate.matchedServices.some((s) => s.categoryId === context.category!.id)
   615→    serviceScore = { ...serviceScore, score: exact ? SERVICE_MATCH_SCORES.exact : SERVICE_MATCH_SCORES.ancestor }
   616→    if (exact) reasons.push(`Offers ${context.category.name}`)
   617→  }
   618→
   619→  // LOCATION / SERVICE AREA MATCH -------------------------------------------
   620→  const exactTownId =
   621→    context.location.scope.kind === 'town'
   622→      ? context.location.scope.townId
   623→      : context.location.contextTownId
   624→  const exactLocation =
   625→    exactTownId !== null &&
   626→    (candidate.primaryLocation?.id === exactTownId ||
   627→      candidate.serviceAreas.some((area) => area.location.id === exactTownId))
   628→
   629→  let distanceKm: number | null = null
   630→  if (context.location.point) {
   631→    const towns = [
   632→      candidate.primaryLocation,
   633→      ...candidate.serviceAreas.map((a) => a.location),
   634→    ].filter((town): town is CandidateTown => town !== null)
   635→    let best: number | null = null
   636→    for (const town of towns) {
   637→      const townPoint = pointOf(town.latitude, town.longitude)
   638→      if (!townPoint) continue
   639→      const d = haversineKmLocal(context.location.point, townPoint)
   640→      if (best === null || d < best) best = d
   641→    }
   642→    distanceKm = best
   643→  }
   644→
   645→  let locationScore: number
   646→  if (exactLocation) {
   647→    locationScore = LOCATION_HIERARCHY_SCORES.communityOrTown
   648→    reasons.push('Serves your area')
   649→  } else if (distanceKm !== null) {
   650→    const band = distanceBandFor(distanceKm)
   651→    locationScore = band.score
   652→    reasons.push(`About ${band.label} away`)
   653→  } else {
   654→    locationScore =
   655→      context.location.scope.kind === 'district'
   656→        ? LOCATION_HIERARCHY_SCORES.districtSweep
   657→        : LOCATION_HIERARCHY_SCORES.regionSweep
   658→    reasons.push('Serves your wider area')
   659→  }
   660→  const locationComponent: ScoredComponent = {
   661→    key: 'locationMatch',
   662→    score: locationScore,
   663→    available: context.location.scope.kind !== 'none' || context.location.point !== null,
   664→  }
   665→
   666→  // AVAILABILITY (real field, PART 13 — never inferred from account status) --
   667→  const availabilityScore = AVAILABILITY_SCORES[candidate.availabilityStatus] ?? NEUTRAL_COMPONENT_SCORE
   668→  if (candidate.availabilityStatus === 'AVAILABLE') reasons.push('Available now')
   669→  const availabilityComponent: ScoredComponent = {
   670→    key: 'availability',
   671→    score: availabilityScore,
   672→    available: true,
   673→  }
   674→
   675→  // RATING / REVIEW QUALITY --------------------------------------------------
   676→  const hasRating = candidate.ratingCount >= RATING_MIN_COUNT
   677→  const ratingAverage = hasRating ? candidate.ratingSum / candidate.ratingCount : null
   678→  const ratingComponent: ScoredComponent = {
   679→    key: 'rating',
   680→    score: hasRating ? Math.min(Math.max(ratingAverage! / 5, 0), 1) : NEUTRAL_COMPONENT_SCORE,
   681→    available: hasRating, // unrated providers are NOT falsely scored
   682→  }
   683→  if (hasRating && ratingAverage! >= 4) reasons.push('Highly rated')
   684→
   685→  // RESPONSE RATE ------------------------------------------------------------
   686→  const responseComponent: ScoredComponent = {
   687→    key: 'responseRate',
   688→    score:
   689→      candidate.responseRatePercent === null
   690→        ? NEUTRAL_COMPONENT_SCORE
   691→        : Math.min(Math.max(candidate.responseRatePercent / 100, 0), 1),
   692→    available: candidate.responseRatePercent !== null,
   693→  }
   694→
   695→  // EXPERIENCE ---------------------------------------------------------------
   696→  const experienceComponent: ScoredComponent = {
   697→    key: 'experience',
   698→    score: Math.min(candidate.yearsExperience / EXPERIENCE_FULL_SCORE_YEARS, 1),
   699→    available: candidate.yearsExperience > 0,
   700→  }
   701→
   702→  // VERIFICATION -------------------------------------------------------------
   703→  const verificationScore = VERIFICATION_SCORES[candidate.verificationStatus] ?? 0
   704→  if (candidate.verificationStatus === 'VERIFIED') reasons.push('Verified provider')
   705→  const verificationComponent: ScoredComponent = {
   706→    key: 'verification',
   707→    score: verificationScore,
   708→    available: true,
   709→  }
   710→
   711→  const components: Record<string, ScoredComponent> = {
   712→    serviceMatch: serviceScore,
   713→    locationMatch: locationComponent,
   714→    availability: availabilityComponent,
   715→    rating: ratingComponent,
   716→    responseRate: responseComponent,
   717→    experience: experienceComponent,
   718→    verification: verificationComponent,
   719→  }
   720→
   721→  // Neutral-fill: components without real data score NEUTRAL (0.5) and keep
   722→  // their weight — missing information pulls toward the middle, never boosts
   723→  // an empty profile above a substantive one, and never fabricates data.
   724→  let weighted = 0
   725→  for (const component of Object.values(components)) {
   726→    const score = component.available ? component.score : NEUTRAL_COMPONENT_SCORE
   727→    weighted += DISCOVERY_WEIGHTS[component.key] * score
   728→  }
   729→  const total = weighted * 100
   730→
   731→  return {
   732→    candidate,
   733→    components,
   734→    total,
   735→    distanceKm,
   736→    distanceBand: distanceKm !== null ? distanceBandFor(distanceKm) : null,
   737→    exactLocation,
   738→    reasons: [...new Set(reasons)],
   739→  }
   740→}
   741→
   742→// -----------------------------------------------------------------------------
   743→// Sorting (PART 15 — unavailable information never falsely ranks)
   744→// -----------------------------------------------------------------------------
   745→
   746→function sortScored(scored: ScoredCandidate[], sort: DiscoveryQuery['sort']): ScoredCandidate[] {
   747→  const byScore = (a: ScoredCandidate, b: ScoredCandidate) => b.total - a.total || a.candidate.id.localeCompare(b.candidate.id)
   748→  const list = [...scored]
   749→
   750→  switch (sort) {
   751→    case 'nearest':
   752→      return list.sort((a, b) => {
   753→        if (a.distanceKm === null && b.distanceKm === null) return byScore(a, b)
   754→        if (a.distanceKm === null) return 1
   755→        if (b.distanceKm === null) return -1
   756→        return a.distanceKm - b.distanceKm || byScore(a, b)
   757→      })
   758→    case 'rating':
   759→      return list.sort((a, b) => {
   760→        const aRated = a.candidate.ratingCount >= RATING_MIN_COUNT
   761→        const bRated = b.candidate.ratingCount >= RATING_MIN_COUNT
   762→        if (aRated !== bRated) return aRated ? -1 : 1
   763→        if (aRated && bRated) {
   764→          const aAvg = a.candidate.ratingSum / a.candidate.ratingCount
   765→          const bAvg = b.candidate.ratingSum / b.candidate.ratingCount
   766→          if (bAvg !== aAvg) return bAvg - aAvg
   767→        }
   768→        return byScore(a, b)
   769→      })
   770→    case 'experience':
   771→      return list.sort(
   772→        (a, b) => b.candidate.yearsExperience - a.candidate.yearsExperience || byScore(a, b),
   773→      )
   774→    case 'price':
   775→      return list.sort((a, b) => {
   776→        if (a.candidate.startingPriceAmount === null && b.candidate.startingPriceAmount === null) return byScore(a, b)
   777→        if (a.candidate.startingPriceAmount === null) return 1
   778→        if (b.candidate.startingPriceAmount === null) return -1
   779→        return a.candidate.startingPriceAmount - b.candidate.startingPriceAmount || byScore(a, b)
   780→      })
   781→    case 'response':
   782→      return list.sort((a, b) => {
   783→        if (a.candidate.responseRatePercent === null && b.candidate.responseRatePercent === null) return byScore(a, b)
   784→        if (a.candidate.responseRatePercent === null) return 1
   785→        if (b.candidate.responseRatePercent === null) return -1
   786→        return b.candidate.responseRatePercent - a.candidate.responseRatePercent || byScore(a, b)
   787→      })
   788→    default:
   789→      return list.sort(byScore)
   790→  }
   791→}

   793→// -----------------------------------------------------------------------------
   794→// The pipeline (PART 20)
   795→// -----------------------------------------------------------------------------
   796→
   797→export async function discoverProviders(query: DiscoveryQuery): Promise<DiscoveryResponse> {
   798→  const [location, category] = await Promise.all([resolveLocation(query), resolveCategory(query)])
   799→
   800→  // Service search implies the service's own category for scoring purposes.
   801→  let serviceId: string | null = null
   802→  let effectiveCategory = category
   803→  if (query.serviceId) {
   804→    const service = await db.service.findFirst({
   805→      where: { id: query.serviceId, status: 'ACTIVE', deletedAt: null },
   806→      select: { id: true, categoryId: true, category: { select: { id: true, name: true, slug: true } } },
   807→    })
   808→    if (!service) throw new ValidationError({ serviceId: ['Unknown service'] })
   809→    serviceId = service.id
   810→    if (!effectiveCategory) {
   811→      effectiveCategory = {
   812→        id: service.category.id,
   813→        name: service.category.name,
   814→        slug: service.category.slug,
   815→        description: null,
   816→        subtreeIds: [service.category.id],
   817→      }
   818→    }
   819→  }
   820→
   821→  // ---- Hard filters (PART 10) ---------------------------------------------
   822→  // Conditions are composed as an AND list so independent OR-shapes (location
   823→  // scope vs free-text q) can never collide on a top-level `OR` key.
   824→  const baseConditions: Record<string, unknown>[] = [
   825→    // Suspension is a LIVE user-status check (PART 45) — suspended providers
   826→    // never appear publicly, even mid-session.
   827→    { user: { status: 'ACTIVE' } },
   828→  ]
   829→  if (query.verification) baseConditions.push({ verificationStatus: query.verification })
   830→  if (query.availability) baseConditions.push({ availabilityStatus: query.availability })
   831→  if (query.providerType === 'INDIVIDUAL') baseConditions.push({ businessId: null })
   832→  if (query.providerType === 'BUSINESS') baseConditions.push({ businessId: { not: null } })
   833→  if (query.maxPrice !== undefined) {
   834→    baseConditions.push({ startingPriceAmount: { lte: query.maxPrice, not: null } })
   835→  }
   836→  if (effectiveCategory) {
   837→    baseConditions.push({
   838→      services: {
   839→        some: {
   840→          status: 'ACTIVE',
   841→          deletedAt: null,
   842→          isAvailable: true,
   843→          categoryId: { in: effectiveCategory.subtreeIds },
   844→          ...(serviceId ? { id: serviceId } : {}),
   845→        },
   846→      },
   847→    })
   848→  }
   849→  if (query.q) {
   850→    baseConditions.push({
   851→      OR: [
   852→        { profession: { contains: query.q.toLowerCase() } },
   853→        { headline: { contains: query.q } },
   854→        { biography: { contains: query.q } },
   855→        { business: { name: { contains: query.q } } },
   856→      ],
   857→    })
   858→  }
   859→  const locationCondition = locationWhere(location.scope)
   860→  // `where` = base hard filters + the exact location tier. `baseConditions`
   861→  // itself stays location-free: the nearby fallback reuses it with a swapped
   862→  // location condition (requested town → nearby towns).
   863→  const where = {
   864→    deletedAt: null,
   865→    AND:
   866→      Object.keys(locationCondition).length > 0
   867→        ? [...baseConditions, locationCondition]
   868→        : baseConditions,
   869→  }
   870→  const serviceSelect = {
   871→    where: {
   872→      status: 'ACTIVE' as const,
   873→      deletedAt: null as null,
   874→      ...(effectiveCategory ? { categoryId: { in: effectiveCategory.subtreeIds } } : {}),
   875→    },
   876→    select: { name: true, categoryId: true },
   877→    take: 10,
   878→  }
   879→
   880→  // ---- Fetch candidates (single query, capped, minimal selects) ------------
   881→  const rows = await db.providerProfile.findMany({
   882→    where,
   883→    take: DISCOVERY_CANDIDATE_CAP,
   884→    select: { ...CANDIDATE_SELECT, services: serviceSelect },
   885→  })
   886→
   887→  const candidates: Candidate[] = rows.map((row) => {
   888→    const shaped = shapeCandidate(row)
   889→    shaped.matchedServices = row.services
   890→    return shaped
   891→  })
   892→
   893→  // Post-fetch hard filters that SQL cannot express on SQLite (rating).
   894→  const minRating = query.minRating ?? null
   895→  const eligible = candidates.filter((candidate) => {
   896→    if (minRating !== null && minRating > 0) {
   897→      if (candidate.ratingCount < RATING_MIN_COUNT) return false
   898→      if (candidate.ratingSum / candidate.ratingCount < minRating) return false
   899→    }
   900→    return true
   901→  })
   902→
   903→  // ---- Score, filter by distance, sort, paginate ---------------------------
   904→  const scoringContext = { category: effectiveCategory, location, maxDistanceKm: query.maxDistanceKm ?? null }
   905→  let scored = eligible
   906→    .map((candidate) => scoreCandidate(candidate, scoringContext))
   907→    .filter((s) => {
   908→      if (query.maxDistanceKm !== undefined && s.distanceKm !== null) {
   909→        return s.distanceKm <= query.maxDistanceKm
   910→      }
   911→      return true
   912→    })
   913→
   914→  // Duplicate prevention is structural: one row per ProviderProfile.
   915→  const seen = new Set<string>()
   916→  scored = scored.filter((s) => (seen.has(s.candidate.id) ? false : (seen.add(s.candidate.id), true)))
   917→
   918→  const ranked = sortScored(scored, query.sort)
   919→  const total = ranked.length
   920→  const page = query.page
   921→  const pageSize = query.pageSize
   922→  const pageItems = ranked.slice((page - 1) * pageSize, (page - 1) * pageSize + pageSize)
   923→
   924→  // ---- Nearby fallback (PART 18) — only when nothing matched exactly and a
   925→  // real search point exists to measure "nearby" from --------------------------
   926→  let nearby: DiscoveryResponse['nearby'] = null
   927→  if (total === 0 && location.point) {
   928→    const nearbyTowns = await db.town.findMany({
   929→      where: {
   930→        isActive: true,
   931→        latitude: { not: null },
   932→        longitude: { not: null },
   933→      },
   934→      select: { id: true, name: true, latitude: true, longitude: true },
   935→    })
   936→    const within = nearbyTowns
   937→      .filter((town) => {
   938→        const d = haversineKmLocal(
   939→          location.point!,
   940→          { latitude: town.latitude as number, longitude: town.longitude as number },
   941→        )
   942→        return d <= NEARBY_RADIUS_KM && d > 0.5
   943→      })
   944→      .map((town) => ({
   945→        town,
   946→        distanceKm: haversineKmLocal(
   947→          location.point!,
   948→          { latitude: town.latitude as number, longitude: town.longitude as number },
   949→        ),
   950→      }))
   951→      .sort((a, b) => a.distanceKm - b.distanceKm)
   952→      .slice(0, 12)
   953→
   954→    if (within.length > 0) {
   955→      // Nearby keeps EVERY hard filter (service, eligibility, status) but
   956→      // swaps the requested-location condition for the nearby-town set — the
   957→      // customer's original selection is never silently replaced (PART 18).
   958→      const nearbyRows = await db.providerProfile.findMany({
   959→        where: {
   960→          deletedAt: null,
   961→          AND: [
   962→            ...baseConditions,
   963→            { serviceAreas: { some: { locationId: { in: within.map((w) => w.town.id) } } } },
   964→          ],
   965→        },
   966→        take: DISCOVERY_CANDIDATE_CAP,
   967→        select: { ...CANDIDATE_SELECT, services: serviceSelect },
   968→      })
   969→      const distanceByTown = new Map(within.map((w) => [w.town.id, w.distanceKm]))
   970→      const nearbyScored = nearbyRows
   971→        .map((row) => {
   972→          const shaped = shapeCandidate(row)
   973→          shaped.matchedServices = row.services
   974→          const scoredCandidate = scoreCandidate(shaped, scoringContext)
   975→          // Nearest served town drives the nearby label.
   976→          const served = shaped.serviceAreas.find((a) => distanceByTown.has(a.location.id))
   977→          scoredCandidate.distanceKm = served ? (distanceByTown.get(served.location.id) ?? null) : null
   978→          scoredCandidate.distanceBand = scoredCandidate.distanceKm !== null ? distanceBandFor(scoredCandidate.distanceKm) : null
   979→          scoredCandidate.reasons = [
   980→            ...scoredCandidate.reasons.filter((r) => !r.startsWith('About ') && !r.startsWith('Serves')),
   981→            served ? `Serves ${served.location.name}` : 'Nearby provider',
   982→          ]
   983→          return scoredCandidate
   984→        })
   985→        .sort((a, b) => (a.distanceKm ?? Infinity) - (b.distanceKm ?? Infinity) || b.total - a.total)
   986→      nearby = {
   987→        label: 'Nearby providers',
   988→        items: nearbyScored.slice(0, pageSize).map(toResult),
   989→        total: nearbyScored.length,
   990→      }
   991→    }
   992→  }
   993→
   994→  // ---- Location context for breadcrumbs / empty states ---------------------
   995→  const nearbyAreas = await resolveNearbyAreas(location)
   996→  const locationContext: DiscoveryLocationContext = {
   997→    level: location.level,
   998→    label: location.label,
   999→    breadcrumb: location.breadcrumb,
  1000→    nearbyAreas,
  1001→  }
  1002→
  1003→  return {
  1004→    criteria: {
  1005→      categoryId: effectiveCategory?.id ?? null,
  1006→      categoryName: effectiveCategory?.name ?? null,
  1007→      categorySlug: effectiveCategory?.slug ?? null,
  1008→      serviceId,
  1009→      regionId: query.regionId ?? null,
  1010→      districtId: query.districtId ?? null,
  1011→      townId: query.townId ?? null,
  1012→      communityId: query.communityId ?? null,
  1013→      providerType: query.providerType ?? null,
  1014→      verification: query.verification ?? null,
  1015→      availability: query.availability ?? null,
  1016→      minRating: query.minRating ?? null,
  1017→      maxPrice: query.maxPrice ?? null,
  1018→      q: query.q ?? null,
  1019→      sort: query.sort,
  1020→      whatLabel: effectiveCategory?.name ?? (query.q ? `“${query.q}”` : 'Providers'),
  1021→    },
  1022→    location: locationContext,
  1023→    items: pageItems.map(toResult),
  1024→    pagination: buildPaginationMeta(total, page, pageSize),
  1025→    nearby,
  1026→  }
  1027→}

  1029→/** Suggested nearby areas for empty states: same-district towns, then towns
  1030→ *  within the nearby radius, then nothing (the UI hides the block). */
  1031→async function resolveNearbyAreas(location: ResolvedLocation): Promise<DiscoveryLocationContext['nearbyAreas']> {
  1032→  if (location.scope.kind === 'town') {
  1033→    const town = await db.town.findFirst({
  1034→      where: { id: location.scope.townId },
  1035→      select: { districtId: true, id: true, latitude: true, longitude: true },
  1036→    })
  1037→    if (!town) return []
  1038→
  1039→    if (town.latitude !== null && town.longitude !== null) {
  1040→      const point = { latitude: town.latitude, longitude: town.longitude }
  1041→      const towns = await db.town.findMany({
  1042→        where: { isActive: true, id: { not: town.id }, latitude: { not: null }, longitude: { not: null } },
  1043→        select: { id: true, name: true, slug: true, latitude: true, longitude: true, isMajor: true },
  1044→        orderBy: [{ isMajor: 'desc' }, { name: 'asc' }],
  1045→      })
  1046→      return towns
  1047→        .map((t) => ({
  1048→          id: t.id,
  1049→          name: t.name,
  1050→          slug: t.slug,
  1051→          isMajor: t.isMajor,
  1052→          distanceKm: haversineKmLocal(point, {
  1053→            latitude: t.latitude as number,
  1054→            longitude: t.longitude as number,
  1055→          }),
  1056→        }))
  1057→        .filter((t) => t.distanceKm <= NEARBY_RADIUS_KM)
  1058→        .sort((a, b) => a.distanceKm - b.distanceKm)
  1059→        .slice(0, 6)
  1060→        .map(({ id, name, slug, distanceKm }) => ({ id, name, slug, distanceKm }))
  1061→    }
  1062→
  1063→    const siblings = await db.town.findMany({
  1064→      where: { districtId: town.districtId, isActive: true, id: { not: town.id } },
  1065→      select: { id: true, name: true, slug: true },
  1066→      orderBy: { name: 'asc' },
  1067→      take: 6,
  1068→    })
  1069→    return siblings.map((t) => ({ ...t, distanceKm: null }))
  1070→  }
  1071→
  1072→  if (location.scope.kind === 'district') {
  1073→    const towns = await db.town.findMany({
  1074→      where: { districtId: location.scope.districtId, isActive: true },
  1075→      select: { id: true, name: true, slug: true },
  1076→      orderBy: [{ isMajor: 'desc' }, { name: 'asc' }],
  1077→      take: 6,
  1078→    })
  1079→    return towns.map((t) => ({ ...t, distanceKm: null }))
  1080→  }
  1081→
  1082→  return []
  1083→}

  1085→/** Maps a scored candidate to the public result shape (privacy final gate). */
  1086→function toResult(scored: ScoredCandidate): DiscoveryResult {
  1087→  const c = scored.candidate
  1088→  const rating =
  1089→    c.ratingCount >= RATING_MIN_COUNT
  1090→      ? { average: Math.round((c.ratingSum / c.ratingCount) * 10) / 10, count: c.ratingCount }
  1091→      : null
  1092→
  1093→  const areas = c.serviceAreas.map(({ location }) => ({
  1094→    id: location.id,
  1095→    name: location.name,
  1096→    regionName: location.regionName,
  1097→  }))
  1098→  const base = c.primaryLocation
  1099→    ? [{ id: c.primaryLocation.id, name: c.primaryLocation.name, regionName: c.primaryLocation.regionName }]
  1100→    : []
  1101→  const merged = [...base]
  1102→  for (const area of areas) {
  1103→    if (merged.length >= 5) break
  1104→    if (!merged.some((m) => m.id === area.id)) merged.push(area)
  1105→  }
  1106→
  1107→  return {
  1108→    providerId: c.id,
  1109→    displayName: c.business?.name ?? c.user.name ?? 'Dwellers provider',
  1110→    providerType: c.businessId ? 'BUSINESS' : 'INDIVIDUAL',
  1111→    businessId: c.businessId,
  1112→    businessName: c.business?.name ?? null,
  1113→    profession: c.profession,
  1114→    headline: c.headline,
  1115→    avatarKey: c.user.avatarKey,
  1116→    verificationStatus: c.verificationStatus,
  1117→    rating,
  1118→    yearsExperience: c.yearsExperience,
