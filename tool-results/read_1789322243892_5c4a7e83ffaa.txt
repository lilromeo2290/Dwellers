     2→     2→ * Dwellers — Projects module: job requests (Phase 5, REQUEST SERVICE).
     3→     3→ *
     4→     4→ * The heart of Dwellers: a customer describes what they need ("my kitchen
     5→     5→ * sink is leaking — I need a plumber in Nsawam"), attaches photos if helpful,
     6→     6→ * picks the job location and submits — no phone call required. The targeted

    14→    14→ *    (for business-affiliated profiles) an ACTIVE OWNER/MANAGER business

    16→    16→ *  - READ: customers see their OWN requests; the targeted provider side sees
    17→    17→ *    requests addressed to them; staff sees all. Cross-user reads are

    20→    20→ * State changes are validated by the central machine, executed inside
    21→    21→ * transactions together with their timeline event, and never trust the

   190→   190→/**
   191→   191→ * Loads a request and resolves the caller's relationship to it. Foreign
   192→   192→ * callers receive the same opaque 404 as anonymous probes — the existence of
   193→   193→ * a request is never disclosed (PART 49/52).
   194→   194→ */
   195→   195→async function resolveAccess(auth: AuthContext | null, id: string): Promise<RequestAccess> {
   196→   196→  const context = requireAuth(auth)
   197→   197→  const request = await db.jobRequest.findUnique({
   198→   198→    where: { id },
   199→   199→    select: {
   200→   200→      id: true,
   201→   201→      reference: true,
   202→   202→      customerId: true,
   203→   203→      providerId: true,
   204→   204→      serviceId: true,
   205→   205→      locationId: true,
   206→   206→      communityId: true,
   207→   207→      status: true,
   208→   208→      viewedAt: true,
   209→   209→    },
   210→   210→  })
   211→   211→  if (!request) throw new NotFoundError('Job request')
   212→   212→
   213→   213→  if (request.customerId === context.userId) {
   214→   214→    return { request, side: 'CUSTOMER', canManageProviderSide: false }
   215→   215→  }
   216→   216→
   217→   217→  if (isStaff(context.role)) {
   218→   218→    return { request, side: 'STAFF', canManageProviderSide: false }
   219→   219→  }
   220→   220→
   221→   221→  if (request.providerId && ROLE_DEFINITIONS[context.role].isProvider) {
   222→   222→    const provider = await db.providerProfile.findUnique({
   223→   223→      where: { id: request.providerId },
   224→   224→      select: { userId: true, businessId: true },
   225→   225→    })
   226→   226→    if (provider) {
   227→   227→      if (provider.userId === context.userId) {
   228→   228→        return { request, side: 'PROVIDER', canManageProviderSide: true }
   229→   229→      }
   230→   230→      if (provider.businessId) {
   231→   231→        // Business-affiliated profile: OWNER/MANAGER manage; MEMBER read-only
   232→   232→        // (PART 27 — never blanket owner-level powers).
   233→   233→        const membership = await db.businessMember.findUnique({
   234→   234→          where: { businessId_userId: { businessId: provider.businessId, userId: context.userId } },
   235→   235→          select: { memberRole: true, status: true },
   236→   236→        })
   237→   237→        if (membership && membership.status === 'ACTIVE') {
   238→   238→          return {
   239→   239→            request,
   240→   240→            side: 'PROVIDER',
   241→   241→            canManageProviderSide: membership.memberRole === 'OWNER' || membership.memberRole === 'MANAGER',
   242→   242→          }
   243→   243→        }
   244→   244→      }
   245→   245→    }
   246→   246→  }
   247→   247→
   248→   248→  throw new NotFoundError('Job request')
   249→   249→}

   256→   256→function pruneUndefined<T extends Record<string, unknown>>(input: T): T {
   257→   257→  const result: Record<string, unknown> = {}
   258→   258→  for (const [key, value] of Object.entries(input)) {
   259→   259→    if (value !== undefined) result[key] = value
   260→   260→  }
   261→   261→  return result as T

   269→   269→async function validateLocationHierarchy(input: { locationId?: string | null; communityId?: string | null }) {

   446→   446→/**
   447→   447→ * Creates a job request for the CALLING customer (draft or direct submit).
   448→   448→ * Idempotent on clientToken (PART 43): a network retry or double click
   449→   449→ * returns the SAME request instead of creating a duplicate.
   450→   450→ */
   451→   451→export async function createJobRequest(auth: AuthContext | null, input: CreateJobRequestInput) {
   452→   452→  const context = requireAuth(auth)
   453→   453→
   454→   454→  // Idempotency first — a retry with the same token resolves to the
   455→   455→  // original request no matter where the first attempt failed.
   456→   456→  if (input.clientToken) {
   457→   457→    const existing = await db.jobRequest.findFirst({
   458→   458→      where: { customerId: context.userId, clientToken: input.clientToken },
   459→   459→      select: { id: true },
   460→   460→    })
   461→   461→    if (existing) return existing.id
   462→   462→  }
   463→   463→
   464→   464→  await assertJobRequestRefs(input)
   465→   465→  if (input.serviceId && input.providerId) {
   466→   466→    await assertProviderOffersService(input.providerId, input.serviceId)
   467→   467→  }
   468→   468→

   547→   547→      entityId: id,

   549→   549→    })

   561→   561→        select: { id: true },
   562→   562→      })
   563→   563→      if (existing) return existing.id
   564→   564→    }
   565→   565→    throw error
   566→   566→  }
   567→   567→}

   623→   623→    ? { status: query.status }

   626→   626→      : {}

   633→   633→      where: { userId: context.userId },
   634→   634→      select: { id: true, businessId: true },
   635→   635→    })

   738→   738→          },
   739→   739→        }
   740→   740→      : {}),
   741→   741→    ...(servesLocation !== undefined ? { servesLocation } : {}),
   742→   742→  }
   743→   743→  return result
   744→   744→}

   747→   747→function displayNameInitial(name: string | null): string {
   748→   748→  const parts = (name ?? '')
   749→   749→    .trim()
   750→   750→    .split(/\s+/)
   751→   751→    .filter(Boolean)
   752→   752→    // Skip parenthesised suffixes like "(Demo)" — never part of a real name.
   753→   753→    .filter((part) => !/^[(@[]/.test(part))
   754→   754→  if (parts.length === 0) return 'A Dwellers customer'
   755→   755→  const first = parts[0]
   756→   756→  const initial = parts.length > 1 ? ` ${parts[parts.length - 1][0]?.toUpperCase() ?? ''}.` : ''
   757→   757→  return `${first}${initial}`
   758→   758→}

   891→   891→      })

   905→   905→

   907→   907→  assertTransition('edit', 'CUSTOMER', access.request.status)
   908→   908→  if (!EDITABLE_STATUSES.includes(access.request.status as never)) {
   909→   909→    throw new BadRequestError('Only draft requests can be edited. Cancel it and create a new one.')
   910→   910→  }
   911→   911→  await assertJobRequestRefs(input)
   912→   912→  if (input.serviceId && access.request.providerId) {
   913→   913→    await assertProviderOffersService(access.request.providerId, input.serviceId)
   914→   914→  }
   915→   915→
   916→   916→  const { action: _action, cancellationReason: _reason, ...fields } = input
   917→   917→  await db.$transaction(async (tx) => {
   918→   918→    await tx.jobRequest.update({
   919→   919→      where: { id },
   920→   920→      data: fields as Prisma.JobRequestUncheckedUpdateInput,
   921→   921→    })
   922→   922→    if (Object.keys(fields).length > 0) {
   923→   923→      await tx.jobRequestEvent.create({
   924→   924→        data: { jobRequestId: id, eventType: 'EDITED', actorId: context.userId, actorRole: context.role },
   925→   925→      })

   929→   929→    actorId: context.userId,
   930→   930→    actorRole: context.role,
   931→   931→    action: AUDIT_ACTIONS.JOB_REQUEST_UPDATED,
   932→   932→    entityType: 'JobRequest',
   933→   933→    entityId: id,
   934→   934→    metadata: { reference: access.request.reference },
   935→   935→  })
   936→   936→  const request = await db.jobRequest.findUnique({ where: { id }, ...listSelect })
   937→   937→  return withListCount(request!)
   938→   938→}
   939→   939→
   940→   940→// -----------------------------------------------------------------------------
   941→   941→// Provider response (PART 31/32)
   942→   942→// -----------------------------------------------------------------------------
   943→   943→
   944→   944→/**
   945→   945→ * The targeted provider responds: express interest, ask for more
   946→   946→ * information, or decline — with an optional short message. The response is
   947→   947→ * atomic: status + responseKind + timeline event + customer notification.
   948→   948→ */
   949→   949→export async function respondToJobRequest(
   950→   950→  auth: AuthContext | null,
   951→   951→  id: string,
   952→   952→  input: RespondToJobRequestInput,
   953→   953→) {
   954→   954→  const context = requireAuth(auth)
   955→   955→  const access = await resolveAccess(auth, id)
   956→   956→  if (access.side !== 'PROVIDER' || !access.canManageProviderSide) {
   957→   957→    throw new ForbiddenError('Only the provider this request was sent to can respond.')
   958→   958→  }
   959→   959→
   960→   960→  // Central machine — provider actions only from waiting states.
   961→   961→  assertTransition(input.action, 'PROVIDER', access.request.status)
   962→   962→  const transition = RESPONSE_TRANSITIONS[input.action]
   963→   963→
   964→   964→  // Defense in depth: a suspended/deactivated provider can never act on a
   965→   965→  // request even if the HTTP layer were bypassed (PART 75).
   966→   966→  const provider = await db.providerProfile.findUnique({
   967→   967→    where: { id: access.request.providerId! },
   968→   968→    select: { id: true, userId: true, user: { select: { status: true } } },
   969→   969→  })
   970→   970→  if (!provider || provider.user.status !== 'ACTIVE') {
   971→   971→    throw new ForbiddenError('This account can no longer be used. Contact Dwellers support.')
   972→   972→  }
   973→   973→
   974→   974→  await db.$transaction(async (tx) => {
   975→   975→    await tx.jobRequest.update({
   976→   976→      where: { id },
   977→   977→      data: {
   978→   978→        status: transition.status,
   979→   979→        responseKind: transition.kind,
   980→   980→        respondedAt: new Date(),
   981→   981→        viewedAt: access.request.viewedAt ?? new Date(),
   982→   982→      },
   983→   983→    })
   984→   984→    await tx.jobRequestEvent.create({
   985→   985→      data: {
   986→   986→        jobRequestId: id,
   987→   987→        eventType: transition.eventType,
   988→   988→        actorId: context.userId,
   989→   989→        actorRole: context.role,
   990→   990→        message: input.message ?? undefined,
   991→   991→      },
   992→   992→    })
   993→   993→    await tx.notification.create({
   994→   994→      data: {
   995→   995→        recipientId: access.request.customerId,
   996→   996→        type: NOTIFICATION_TYPES.JOB_REQUEST_RESPONSE,
   997→   997→        channel: 'IN_APP',
   998→   998→        title: RESPONSE_NOTIFICATION_TITLES[transition.kind],
   999→   999→        body: 'Open My Job Requests to see the full response.',
  1000→  1000→        entityType: 'JobRequest',
  1001→  1001→        entityId: id,
  1002→  1002→      },
  1003→  1003→      select: { id: true },
  1004→  1004→    })
  1005→  1005→  })
  1006→  1006→
  1007→  1007→  recordAudit({
  1008→  1008→    actorId: context.userId,
  1009→  1009→    actorRole: context.role,
  1010→  1010→    action:
  1011→  1011→      transition.kind === 'DECLINED' ? AUDIT_ACTIONS.JOB_REQUEST_DECLINED : AUDIT_ACTIONS.JOB_REQUEST_RESPONDED,
  1012→  1012→    entityType: 'JobRequest',
  1013→  1013→    entityId: id,
  1014→  1014→    metadata: { reference: access.request.reference, kind: transition.kind },
  1015→  1015→  })
  1016→  1016→  recordDiscoveryEvent({ type: 'provider_response' })
  1017→  1017→
  1018→  1018→  const request = await db.jobRequest.findUnique({ where: { id }, ...listSelect })
  1019→  1019→  return withListCount(request!)
  1020→  1020→}

  1023→  1023→  INTERESTED: 'Your service request has been accepted.',
  1024→  1024→  NEEDS_INFO: 'Your service request needs more information.',
  1025→  1025→  DECLINED: 'Your service request was declined.',
  1026→  1026→}

  1032→  1032→/** The customer manages photos until the provider acts; never afterwards. */
  1033→  1033→async function assertAttachmentsEditable(requestId: string) {
  1034→  1034→  const request = await db.jobRequest.findUnique({
  1035→  1035→    where: { id: requestId },
  1036→  1036→    select: { status: true },
  1037→  1037→  })
  1038→  1038→  if (!request) throw new NotFoundError('Job request')
  1039→  1039→  if (!ATTACHMENT_EDITABLE_STATUSES.includes(request.status as never)) {
  1040→  1040→    throw new BadRequestError('Photos can no longer be changed on this request.')
  1041→  1041→  }
  1042→  1042→  return request.status
  1043→  1043→}

  1045→  1045→export async function recordJobRequestAttachment(
  1046→  1046→  auth: AuthContext | null,
  1047→  1047→  requestId: string,
  1048→  1048→  photo: { storageKey: string; originalName?: string | null; mimeType?: string | null; sizeBytes?: number | null },
  1049→  1049→) {
  1050→  1050→  const context = requireAuth(auth)
  1051→  1051→  const access = await resolveAccess(auth, requestId)
  1052→  1052→  if (access.side !== 'CUSTOMER') {
  1053→  1053→    throw new ForbiddenError('Only the customer who created this request can add photos.')
  1054→  1054→  }
  1055→  1055→  await assertAttachmentsEditable(requestId)
  1056→  1056→
  1057→  1057→  const count = await db.jobRequestAttachment.count({ where: { jobRequestId: requestId } })
  1058→  1058→  if (count >= MAX_JOB_REQUEST_ATTACHMENTS) {
  1059→  1059→    throw new ValidationError({
  1060→  1060→      attachments: [`A request can hold at most ${MAX_JOB_REQUEST_ATTACHMENTS} photos.`],
  1061→  1061→    })
  1062→  1062→  }
  1063→  1063→
  1064→  1064→  const created = await db.$transaction(async (tx) => {
  1065→  1065→    const row = await tx.jobRequestAttachment.create({
  1066→  1066→      data: {
  1067→  1067→        jobRequestId: requestId,
  1068→  1068→        storageKey: photo.storageKey,
  1069→  1069→        originalName: photo.originalName ?? null,
  1070→  1070→        mimeType: photo.mimeType ?? null,
  1071→  1071→        sizeBytes: photo.sizeBytes ?? null,
  1072→  1072→        kind: 'PHOTO',
  1073→  1073→        uploadedById: context.userId,
  1074→  1074→      },
  1075→  1075→    })
  1076→  1076→    await tx.jobRequestEvent.create({
  1077→  1077→      data: {
  1078→  1078→        jobRequestId: requestId,
  1079→  1079→        eventType: 'ATTACHMENT_ADDED',
  1080→  1080→        actorId: context.userId,
  1081→  1081→        actorRole: context.role,
  1082→  1082→        message: row.originalName ?? undefined,
  1083→  1083→      },
  1084→  1084→    })
  1085→  1085→    return row
  1086→  1086→  })
  1087→  1087→
  1088→  1088→  recordAudit({
  1089→  1089→    actorId: context.userId,
  1090→  1090→    actorRole: context.role,
  1091→  1091→    action: AUDIT_ACTIONS.JOB_REQUEST_ATTACHMENT_ADDED,
  1092→  1092→    entityType: 'JobRequest',
  1093→  1093→    entityId: requestId,
  1094→  1094→    metadata: { attachmentId: created.id },
  1095→  1095→  })
  1096→  1096→  recordDiscoveryEvent({ type: 'request_attachment_added' })
  1097→  1097→  return created
  1098→  1098→}
